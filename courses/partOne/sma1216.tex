% COURSE SYNOPSISDESCRIPTION
\subsection*{Course Synopsis}
This course aims at providing the student with the basic concepts of Functions of Several Variables, Matrices and Ordinary Differential Equations which are motivated by Radiography and Engineering Sciences applications while emphasizing the informed use of modern numerical analysis methods.
The objective of this course, therefore, is to present mathematical theories that Engineering Scientists have successfully used so far in a manner that highlights the historical and intellectual connections between the applications in Mathematics, Physics and Engineering.
\vspace{2ex} \hrule\vspace{2ex}

% ----------------------------------------------------- %
% COURSE TOPICS
\subsection*{Topics}
\begin{outline}
    \1 Functions of Several Variables
    \begin{itemize}
        \item Limits and Continuity
        \item Partial Derivatives and the Chain rule
        \item Tangent planes and normal lines
        \item Extrema and the Lagrange Multipliers
        \item Applications of Functions of Several Variables
    \end{itemize}
    \1 Matrices (Linear Algebra)
    \begin{itemize}
        \item Basic Operations of Matrices
        \item Properties and definitions
        \item Elementary Row Operations (Gauss Elimination)
        \item Systems of Equations
        \item Determinants and inverses
        \item Eigenvalues and eigenvectors
        \item Linear independence
        \item Application of Matrices
    \end{itemize}
    \1 Ordinary Differentail Equations
    \begin{itemize}
        \item First Order Differential Equations
        \item Variable separable
        \item Integrating factor
        \item Exact equations
        \item Homogeneous ODEs
        \item Linear second ODEs with constant coefficients
        \item Variation of Parameters
        \item System of Equations
        \item Applications of ODEs to Physics and Engineering
    \end{itemize}
\end{outline}
